cmake_minimum_required(VERSION 3.5.1)
project(cpadpoc CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenFHE CONFIG REQUIRED)
set(CMAKE_CXX_FLAGS ${OpenFHE_CXX_FLAGS})
include_directories(${OPENMP_INCLUDES} ${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR} ${OPENMP_LIBRARIES})
set(CMAKE_EXE_LINKER_FLAGS ${OpenFHE_EXE_LINKER_FLAGS})
link_libraries(${OpenFHE_SHARED_LIBRARIES})
add_executable(calib calib.cpp)
add_executable(attack attack.cpp)
add_executable(sweep sweep.cpp)
add_executable(diag diag.cpp)
add_executable(floodtest floodtest.cpp)
add_executable(msgtest msgtest.cpp)
add_executable(swktest swktest.cpp)
add_executable(direct direct.cpp)
